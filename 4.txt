
import threading
import time
import random

semaphore = threading.Semaphore(2)


def access_resource(process_id):
    print(
        f"Process {process_id} is waiting...",
        flush=True
    )

    with semaphore:
        print(
            f"Process {process_id} has accessed the resource.",
            flush=True
        )

        time.sleep(random.uniform(1, 2))

        print(
            f"Process {process_id} is releasing the resource.",
            flush=True
        )


threads = []

print("Starting Semaphore Simulation...", flush=True)

for i in range(1, 6):
    t = threading.Thread(
        target=access_resource,
        args=(i,)
    )

    threads.append(t)
    t.start()

for t in threads:
    t.join()

print("Simulation completed.", flush=True)
