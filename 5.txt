
import java.util.concurrent.locks.ReentrantReadWriteLock;

public class osExp5 {

    private static int sharedData = 0;

    private static final ReentrantReadWriteLock lock =
            new ReentrantReadWriteLock();

    static class Reader implements Runnable {

        private final int id;

        Reader(int id) {
            this.id = id;
        }

        public void run() {
            lock.readLock().lock();

            try {
                System.out.println(
                    "Reader " + id + " reads: " + sharedData
                );
            } finally {
                lock.readLock().unlock();
            }
        }
    }

    static class Writer implements Runnable {

        public void run() {
            lock.writeLock().lock();

            try {
                sharedData = 100;

                System.out.println(
                    "Writer updates data to: " + sharedData
                );
            } finally {
                lock.writeLock().unlock();
            }
        }
    }

    public static void main(String[] args)
            throws InterruptedException {

        Thread r1 = new Thread(new Reader(1));
        Thread r2 = new Thread(new Reader(2));
        Thread w1 = new Thread(new Writer());
        Thread r3 = new Thread(new Reader(3));

        r1.start();
        r2.start();
        w1.start();
        r3.start();

        r1.join();
        r2.join();
        w1.join();
        r3.join();

        System.out.println(
            "Final shared data: " + sharedData
        );
    }
}
